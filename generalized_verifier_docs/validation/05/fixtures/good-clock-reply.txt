```
#if !defined(CLOCK_H)
#define CLOCK_H

#include <string>

namespace date_independent {

class clock {
public:
    static clock at(int hour, int minute = 0);
    clock& plus(int minutes);
    clock& minus(int minutes);
    operator std::string() const;
    bool operator==(const clock& rhs) const;

private:
    int minutes_;
    static constexpr int MINUTES_IN_DAY = 24 * 60;
};

bool operator!=(const clock& lhs, const clock& rhs);

}  // namespace date_independent

#endif // CLOCK_H
```

clock.cpp
```
#include "clock.h"

#include <string>

namespace date_independent {

clock::clock(int hour, int minute) : minutes_(hour * 60 + minute) {}

clock clock::at(int hour, int minute) {
    return clock(hour, minute);
}

clock& clock::plus(int minutes) {
    minutes_ += minutes;
    normalize();
    return *this;
}

clock& clock::minus(int minutes) {
    minutes_ -= minutes;
    normalize();
    return *this;
}

void clock::normalize() {
    minutes_ %= MINUTES_IN_DAY;
    if (minutes_ < 0) {
        minutes_ += MINUTES_IN_DAY;
    }
}

clock::operator std::string() const {
    int hours = minutes_ / 60;
    int mins = minutes_ % 60;
    std::string h = std::to_string(hours);
    std::string m = std::to_string(mins);
    if (h.size() == 1) h = "0" + h;
    if (m.size() == 1) m = "0" + m;
    return h + ":" + m;
}

bool clock::operator==(const clock& rhs) const {
    return minutes_ == rhs.minutes_;
}

bool operator!=(const clock& lhs, const clock& rhs) {
    return !(lhs == rhs);
}

}  // namespace date_independent
```
