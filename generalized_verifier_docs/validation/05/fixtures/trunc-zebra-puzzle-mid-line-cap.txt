#include "zebra_puzzle.h"

#include <array>
#include <algorithm>
#include <cassert>
#include <string_view>

namespace zebra_puzzle {

namespace {

constexpr std::array<std::string_view, 5> COLORS = {"Red", "Green", "Ivory", 
"Yellow", "Blue"};
constexpr std::array<std::string_view, 5> NATIONALITIES = {"Englishman", 
"Spaniard", "Ukrainian", "Japanese", "Norwegian"};
constexpr std::array<std::string_view, 5> PETS = {"Dog", "Snail", "Fox", 
"Horse", "Zebra"};
constexpr std::array<std::string_view, 5> DRINKS = {"Coffee", "Tea", "Milk", 
"Orange Juice", "Water"};
constexpr std::array<std::string_view, 5> HOBBIES = {"Dancing", "Painting", 
"Football", "Chess", "Reading"};

constexpr int HOUSE_COUNT = 5;

using Assignment = std::array<std::array<int, 5>, HOUSE_COUNT>;
using Options = std::array<std::array<bool, 5>, HOUSE_COUNT>;

bool isValidAssignment(const Assignment& assignment) {
    for (int house = 0; house < HOUSE_COUNT; ++house) {
        for (int attribute = 0; attribute < 5; ++attribute) {
            if (assignment[house][attribute] < 0 || assignment[house][attribute]
>= 5) {
                return false;
            }
        }
    }
    return true;
}

bool areAllDistinct(const Assignment& assignment) {
    for (int attribute = 0; attribute < 5; ++attribute) {
        std::array<bool, 5> present{};
        for (int house = 0; house < HOUSE_COUNT; ++house) {
            const int value = assignment[house][attribute];
            if (present[value]) {
                return false;
            }
            present[value] = true;
        }
    }
    return true;
}

bool checkSpecificConstraints(const Assignment& assignment) {
    for (int house = 0; house < HOUSE_COUNT; ++house) {
        const int color = assignment[house][0];
        const int nationality = assignment[house][1];
        const int pet = assignment[house][2];
        const int drink = assignment[house][3];
        const int hobby = assignment[house][4];

        // 2. Englishman lives in the red house.
        if (nationality == 0 && color != 0) return false;
        // 3. Spaniard owns the dog.
        if (nationality == 1 && pet != 0) return false;
        // 4. Person in green house drinks coffee.
        if (color == 1 && drink != 0) return false;
        // 5. Ukrainian drinks tea.
        if (nationality == 2 && drink != 1) return false;
        // 7. Snail owner likes to go dancing.
        if (pet == 1 && hobby != 0) return false;
        // 8. Person in yellow house is a painter.
        if (color == 3 && hobby != 1) return false;
        // 9. Person in the middle house drinks milk.
        if (house == 2 && drink != 2) return false;
        // 10. Norwegian lives in the first house.
        if (house == 0 && nationality != 4) return false;
        // 13. Person who plays football drinks orange juice.
        if (hobby == 2 && drink != 3) return false;
        // 14. Japanese person plays chess.
        if (nationality == 3 && hobby != 3) return false;
    }

    // 6. Green house is immediately to the right of the ivory house.
    for (int house = 0; house < HOUSE_COUNT - 1; ++house) {
        if (assignment[house][0] == 2 && assignment[house + 1][0] != 1) return 
false;
    }

    // 11. Person who enjoys reading lives in the house next to the person with 
the fox.
    for (int house = 0; house < HOUSE_COUNT - 1; ++house) {
        if (assignment[house][2] == 2 && assignment[house + 1][2] != 4) return 
false;
        if (assignment[house + 1][2] == 2 && assignment[house][2] != 4) return 
false;
    }

    // 12. Painter's house is next to the house with the horse.
    for (int house = 0; house < HOUSE_COUNT - 1; ++house) {
        if (assignment[house][4] == 1 && assignment[house + 1][2] != 3) return 
false;
        if (assignment[house + 1][4] == 1 && assignment[house][2] != 3) return 
false;
    }

    // 15. Norwegian lives next to the blue house.
    for (int house = 0; house < HOUSE_COUNT - 1; ++house) {
        if (assignment[house][4] == 4 && assignment[house + 1][0] != 4) return 
false;
        if (assignment[house + 1][4] == 4 && assignment[house][0] != 4) return 
false;
    }

    return true;
}

bool solveRecursion(const Assignment& assignment, const Options& options, int 
currentHouse) {
    if (currentHouse == HOUSE_COUNT) {
        return checkSpecificConstraints(assignment) && 
areAllDistinct(assignment);
    }

    for (int attribute = 0; attribute < 5; ++attribute) {
        if (!options[currentHouse][attribute]) continue;

        Assignment nextAssignment = assignment;
        nextAssignment[currentHouse] = {attribute, attribute, attribute, 
attribute, attribute};

        Options nextOptions = options;
        nextOptions[currentHouse][attribute] = false;

        if (solveRecursion(nextAssignment, nextOptions, currentHouse + 1)) {
            return true;
        }
    }
    return false;
}

}  // namespace

Solution solve() {
    Assignment assignment{};
    Options options{};
    for (int house = 0; house < HOUSE_COUNT; ++house) {
        for (int attribute = 0; attribute < 5; ++attribute) {
            options[house][attribute] = true;
        }
    }

    if (solveRecursion(assignment, options, 0)) {
        const int waterDrinkHouse = std::find(assignment.begin(), 
assignment.end(), std::array<int, 5>{4, 4, 4, 4, 4}) - assignment.begin();
        const int zebraPetHouse = std::find(assignment.begin(), 
assignment.end(), std::array<int, 5>{3, 3, 3, 3, 3}) - assignment.begin();

        return Solution{std::string(DRINKS[assignment[waterDrinkHouse][3]]),
                        std::string(NATIONALITIES[assignment[zebraPetHouse][1]])
};
    }
    return Solution{"", ""};
}

}  // namespace zebra_puzzle
