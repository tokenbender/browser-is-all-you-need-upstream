#include "allergies.h"

#include <string>
#include <unordered_set>

namespace allergies {

allergy_test::allergy_test(unsigned int test_result) : score_(test_result) {}

bool allergy_test::is_allergic_to(std::string const& allergen) const {
    static std::unordered_map<std::string, unsigned int> const allergens = {
        {"eggs", 1},
        {"peanuts", 2},
        {"shellfish", 4},
        {"strawberries", 8},
        {"tomatoes", 16},
        {"chocolate", 32},
        {"pollen", 64},
        {"cats", 128}
    };

    auto const found = allergens.find(allergen);
    if (found == allergens.end()) {
        return false;
    }
    return (score_ & found->second) != 0;
}

std::unordered_set<std::string> allergy_test::get_allergies() const {
    std::unordered_set<std::string> result;

    static std::unordered_map<std::string, unsigned int> const allergens = {
        {"eggs", 1},
        {"peanuts", 2},
        {"shellfish", 4},
        {"strawberries", 8},
        {"tomatoes", 16},
        {"chocolate", 32},
        {"pollen", 64},
        {"cats", 128}
    };

    for (auto const& [name, value] : allergens) {
        if ((score_ & value) != 0) {
            result.emplace(name);
        }
    }
    return result;
}

}  // namespace allergies
