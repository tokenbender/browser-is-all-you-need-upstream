    #include "crypto_square.h"

    #include <cctype>
    #include <sstream>
    #include <utility>
    #include <vector>

    namespace crypto_square {

    cipher::cipher(std::string const& text) {
        std::stringstream ss(text);
        char c;
        while (ss.get(c)) {
            if (std::isalnum(c)) {
                normalized_ += static_cast<char>(std::tolower(c));
            }
        }
        if (normalized_.empty()) {
            cols_ = 0;
            rows_ = 0;
            return;
        }
        std::size_t length = normalized_.size();
        cols_ = static_cast<std::size_t>(std::ceil(std::sqrt(length)));
        rows_ = (length + cols_ - 1) / cols_;
    }

    std::string cipher::normalize_plain_text() const {
        return normalized_;
    }

    std::size_t cipher::size() const {
        return cols_;
    }

    std::vector<std::string> cipher::plain_text_segments() const {
        std::vector<std::string> segments;
        if (normalized_.empty()) {
            return segments;
        }
        for (std::size_t i = 0; i < rows_; ++i) {
            std::size_t start = i * cols_;
            std::size_t end = std::min(start + cols_, normalized_.size());
            segments.push_back(normalized_.substr(start, end - start));
        }
        return segments;
    }

    std::string cipher::cipher_text() const {
        std::string result;
        if (normalized_.empty()) {
            return result;
        }
        for (std::size_t col = 0; col < cols_; ++col) {
            for (std::size_t row = 0; row < rows_; ++row) {
                result += normalized_[row * cols_ + col];
            }
        }
        return result;
    }

    std::string cipher::normalized_cipher_text() const {
        std::string result;
        if (normalized_.empty()) {
            return result;
        }
        for (std::size_t row = 0; row < rows_; ++row) {
            std::size_t start = row * cols_;
            std::size_t end = std::min(start + cols_, normalized_.size());
            result += normalized_.substr(start, end - start);
            if (row != rows_ - 1) {
                result += ' ';
            }
        }
        return result;
    }

    }  // namespace crypto_square

        }

        std::string result;

            std::size_t end = std::min(start + cols_, normalized_.size());

            }
