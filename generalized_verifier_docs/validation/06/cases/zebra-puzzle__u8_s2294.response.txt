                    for (int d = 0; d < 5; ++d) {
                        if (used_drink[d]) continue;
                        for (int h = 0; h < 5; ++h) {
                            if (used_hobby[h]) continue;

                            if (assign(house, current_colors[c], current_nationalities[n],
                                       current_pets[p], current_drinks[d], current_hobbies[h])) {
                                backtrack(house + 1);
                                used_color[house] = false;
                                used_nationality[house] = false;
                                used_pet[house] = false;
                                used_drink[house] = false;
                                used_hobby[house] = false;
                            }
                        }
                    }
                }
            }
        }
    };

    backtrack(0);
    return Solution();
}

}  // namespace zebra_puzzle
