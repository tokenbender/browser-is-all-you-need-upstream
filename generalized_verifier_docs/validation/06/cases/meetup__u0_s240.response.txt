```cpp
#include "meetup.h"
#include <algorithm>
#include <cassert>
#include <ctime>
#include <iomanip>
#include <sstream>
#include <stdexcept>

namespace meetup {

std::tm meetup_day_of_month(int month, int year, int weekday, int week) {
    std::tm t = {};
    t.tm_year = year - 1900;
    t.tm_mon = month - 1;
    t.tm_mday = 1;
    t.tm_hour = 0;
    t.tm_min = 0;
    t.tm_sec = 0;
    t.tm_isdst = -1;

    if (mktime(&t) == -1) {
        throw std::invalid_argument("Invalid date");
    }

    int day_of_week = t.tm_wday;
    int days_in_month = t.tm_mday;
    t.tm_mday = days_in_month;
    if (mktime(&t) == -1) {
        throw std::invalid_argument("Invalid date");
    }
    days_in_month = t.tm_mday;

    int result_day = 0;

    switch (week) {
        case first:
            // Find the first weekday
            result_day = weekday - day_of_week;
            if (result_day < 0) {
                result_day += 7;
            }
            break;

        case second:
            result_day = 7 + weekday - day_of_week;
            if (result_day < 7) {
                result_day += 7;
            }
            break;

        case third:
            result_day = 14 + weekday - day_of_week;
            if (result_day < 14) {
                result_day += 7;
            }
            break;

        case fourth:
            result_day = 21 + weekday - day_of_week;
            if (result_day < 21) {
                result_day += 7;
            }
            break;

        case teenth: {
            // Days from 13th to 19th
            int start = std::max(13, day_of_week);
            int end = std::min(19, day_of_week + 7);

            for (int d = start; d <= end; ++d) {
                if ((d - day_of_week + 7) % 7 == weekday) {
                    result_day = d;
                    break;
                }
            }
            if (result_day == 0) {
                throw std::out_of_range("No teenth weekday exists");
            }
            break;
        }

        case last: {
            int days_remaining = days_in_month - day_of_week;
            result_day = days_remaining + weekday;
            break;
        }

        default:
            throw std::invalid_argument("Invalid week");
    }

    t.tm_mday = result_day;
    if (mktime(&t) == -1) {
        throw std::out_of_range("Calculated date out of range");
    }

    return t;
}

}  // namespace meetup
```
