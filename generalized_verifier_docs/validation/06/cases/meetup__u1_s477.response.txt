Date meetup_day(int month, int year, Day day, Week week) {
    int target_weekday = static_cast<int>(day);
    std::tm t = {0};
    t.tm_year = year - 1900;
    t.tm_mon = month - 1;
    t.tm_mday = 1;

    std::mktime(&t);
    int days_in_month = t.tm_mday; // tm_mday is set to 0 after mktime, but days in month is stored somehow?
    // Actually, standard way: tm_mday is reset to 1. To get days in month, we can try setting day to 32.
    // Or just use standard library limits if available, but simple math is safer.
    // Let's use the "set to 32" trick.
    t.tm_mday = 32;
    std::mktime(&t);
    days_in_month = t.tm_mday - 1;

    // Helper lambda to check weekday
    auto is_target = [&](int d) -> bool {
        std::tm tm_day = t;
        tm_day.tm_mday = d;
        std::mktime(&tm_day);
        return tm_day.tm_wday == target_weekday;
    };

    int day_of_month;

    if (week == Week::Teenth) {
        for (int d = 13; d <= 19; ++d) {
            if (is_target(d)) {
                day_of_month = d;
                break;
            }
        }
    } else if (week == Week::Last) {
        for (int d = days_in_month; d >= 1; --d) {
            if (is_target(d)) {
                day_of_month = d;
                break;
            }
        }
    } else {
        // First, Second, Third, Fourth

        std::tm tm_first_day = t;
        tm_first_day.tm_mday = 1;
        std::mktime(&tm_first_day);

        int target_first_day_of_month = (target_weekday + 7 - first_target_day) % 7;
        int base_day = target_first_day_of_month + 1; // 1-based

        int week_index = static_cast<int>(week);
        day_of_month = base_day + (week_index - 1) * 7;

        // Safety check (shouldn't happen with valid input, but good practice)
        if (day_of_month < 1 || day_of_month > days_in_month) {
            throw std::invalid_argument("Invalid week or month");
        }
    }

    return {year, month, day_of_month};
}

    if (week == Week::Teenth) {
        for (int d = 13; d <= 19; ++d) {
            if (is_target(d)) {
                day_of_month = d;
                break;
            }
        }
    } else if (week == Week::Last) {
        for (int d = days_in_month; d >= 1; --d) {
            if (is_target(d)) {
                day_of_month = d;
                break;
            }
        }
    } else {
        // First, Second, Third, Fourth
        // Find first day of month that is target weekday
        std::tm tm_first_day = t;
        tm_first_day.tm_mday = 1;
        std::mktime(&tm_first_day);
        int first_target_day = tm_first_day.tm_wday;
        int target_first_day_of_month = (target_weekday + 7 - first_target_day) % 7;
        int base_day = target_first_day_of_month + 1; // 1-based

        int week_index = static_cast<int>(week);
        day_of_month = base_day + (week_index - 1) * 7;

        // Safety check (shouldn't happen with valid input, but good practice)
        if (day_of_month < 1 || day_of_month > days_in_month) {
            throw std::invalid_argument("Invalid week or month");
        }
    }

    return {year, month, day_of_month};
}

}  // namespace meetup
