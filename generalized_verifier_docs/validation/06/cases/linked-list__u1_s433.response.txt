    while (!empty()) {
        pop_front();

void List::push_front(int value) {
    Node* new_node = new Node{value, nullptr, head_};
    if (head_ == nullptr) {
        head_ = tail_ = new_node;
    } else {
        head_->prev = new_node;
        head_ = new_node;
    }
    ++size_;
}

void List::push_back(int value) {
    Node* new_node = new Node{value, tail_, nullptr};
    if (head_ == nullptr) {
        head_ = tail_ = new_node;
    } else {
        tail_->next = new_node;
        tail_ = new_node;
    }
    ++size_;
}

void List::pop_front() {
    if (head_ == nullptr) {
        return;
    }
    Node* old_head = head_;
    head_ = head_->next;
    if (head_ != nullptr) {
        head_->prev = nullptr;
    } else {
        tail_ = nullptr;
    }
    delete old_head;
    --size_;
}

void List::pop_back() {
    if (tail_ == nullptr) {
        return;
    }
    Node* old_tail = tail_;
    tail_ = tail_->prev;
    if (tail_ != nullptr) {
        tail_->next = nullptr;
    } else {
        head_ = nullptr;
    }
    delete old_tail;
    --size_;
}

                pop_back();

                current = current->next;
                delete temp;
            }
            --size_;
            continue;
        }
        current = current->next;
    }
}

    return size_;
}

}
