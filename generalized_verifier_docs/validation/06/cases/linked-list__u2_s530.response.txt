#include "linked_list.h"
#include <cstddef>
#include <stdexcept>

namespace linked_list {

Route::Route() : head(nullptr), tail(nullptr), size(0) {}

Route::~Route() {
    Node* current = head;
    while (current != nullptr) {
        Node* next = current->next;
        delete current;
        current = next;
    }
}

std::size_t Route::size() const {
    return size;
}

void Route::push_front(int station) {
    Node* newNode = new Node{station, head, nullptr};
    if (head == nullptr) {
        head = newNode;
        tail = newNode;
    } else {
        head->prev = newNode;
        head = newNode;
    }
    ++size;
}

void Route::push_back(int station) {
    Node* newNode = new Node{station, nullptr, tail};
    if (tail == nullptr) {
        head = newNode;
        tail = newNode;
    } else {
        tail->next = newNode;
        tail = newNode;
    }
    ++size;
}

void Route::pop_front() {
    if (head == nullptr) {
        throw std::runtime_error("Route is empty");
    }
    Node* temp = head;
    head = head->next;
    if (head != nullptr) {
        head->prev = nullptr;
    } else {
        tail = nullptr;
    }
    delete temp;
    --size;
}

void Route::pop_back() {
    if (tail == nullptr) {
        throw std::runtime_error("Route is empty");
    }
    Node* temp = tail;
    tail = tail->prev;
    if (tail != nullptr) {
        tail->next = nullptr;
    } else {
        head = nullptr;
    }
    delete temp;
    --size;
}

void Route::insert(int station) {
    // Find the node with the matching data
    Node* current = head;
    while (current != nullptr) {
        if (current->data == station) {
            // Remove this node
            if (current->prev != nullptr) {
                current->prev->next = current->next;
            } else {
                head = current->next;
            }

            if (current->next != nullptr) {
                current->next->prev = current->prev;
            } else {
                tail = current->prev;
            }
            
            delete current;
            --size;
            return;
        }
        current = current->next;
    }
}

}  // namespace linked_list
